function loadScript(src) {
  return new Promise(function (res, rej) {
    var s = document.createElement("script"); s.src = src;
    s.onload = res; s.onerror = function () { rej(new Error("load fail: " + src)); };

function fmt(n) { return n == null ? "-" : Math.round(n).toLocaleString("ko-KR"); }
function esc(s) { return String(s == null ? "" : s).replace(/[&<>"]/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]; }); }
function chgCls(v) { return v == null ? "z" : v > 0 ? "u" : v < 0 ? "d" : "z"; }
function parseChg(t) {
  if (!t) return null;
  var m = String(t).match(/^([▲▼])\s*([\d.]+)%/);
  if (m) return (m[1] === "▲" ? 1 : -1) * parseFloat(m[2]);
  if (/^0\.0%/.test(t)) return 0;
  return null;
}
function chgHtml(t) {

  var v = parseChg(t); return '<span class="chg ' + chgCls(v) + '">' + esc(t) + "</span>";
}

function pinImage(label, color, selected) {
  var w = selected ? 40 : 32, h = selected ? 52 : 42;
  var fs = label.length > 3 ? (selected ? 12 : 10) : (selected ? 16 : 13);
  var r = w / 2 - 2;
  var svg = '<svg xmlns="http://www.w3.org/2000/svg" width="' + w + '" height="' + h + '" viewBox="0 0 ' + w + " " + h + '">' +
    '<path d="M' + w / 2 + " " + (h - 1) + " C" + (w / 2 - 3) + " " + (h - 9) + " 2 " + (w * 0.62) + " 2 " + w / 2 + " A" + r + " " + r + " 0 1 1 " + (w - 2) + " " + w / 2 +
    " C" + (w - 2) + " " + (w * 0.62) + " " + (w / 2 + 3) + " " + (h - 9) + " " + w / 2 + " " + (h - 1) + 'Z" fill="' + color + '" stroke="' + (selected ? "#1d2433" : "#fff") + '" stroke-width="' + (selected ? 2.5 : 1.8) + '"/>' +
    '<text x="' + w / 2 + '" y="' + (w / 2 + fs * 0.36) + '" text-anchor="middle" font-family="Malgun Gothic,Arial,sans-serif" font-weight="700" font-size="' + fs + '" fill="#fff">' + label + "</text></svg>";
  return new kakao.maps.MarkerImage("data:image/svg+xml;charset=utf-8," + encodeURIComponent(svg),

}

function distM(a, b) {
  var R = 6371000, dLa = (b[0] - a[0]) * Math.PI / 180, dLo = (b[1] - a[1]) * Math.PI / 180;
  var x = Math.sin(dLa / 2) ** 2 + Math.cos(a[0] * Math.PI / 180) * Math.cos(b[0] * Math.PI / 180) * Math.sin(dLo / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(x));
}

    d = d.filter(function (p) { return distM(centerLL, [+p.y, +p.x]) <= CONFIG.SEARCH_RADIUS_M; });

  }
  return null;
}

function placeMarker(e) {
  var c = coordOf(e.id) || (e.tmp || null);
  if (!c) return;
  var pos = new kakao.maps.LatLng(c[0], c[1]);
  var m = markers[e.id];
  if (!m) {
    m = new kakao.maps.Marker({ position: pos, image: pinImage(e.label, colorOf(e), false), draggable: false, clickable: true, zIndex: 3 });
    m.setMap(map); markers[e.id] = m; m.__e = e;
    kakao.maps.event.addListener(m, "mouseover", function () {
      tipOv.setContent('<div class="tip">' + tipHtml(e) + "</div>"); tipOv.setPosition(m.getPosition()); tipOv.setMap(map);
    });
    kakao.maps.event.addListener(m, "mouseout", function () { tipOv.setMap(null); });
    kakao.maps.event.addListener(m, "click", function () { if (e.label !== "SITE") openDetail(e.id, false); });
    kakao.maps.event.addListener(m, "dragend", function () {
      var p = m.getPosition(); LOCAL_FIX[e.id] = [+p.getLat().toFixed(6), +p.getLng().toFixed(6)]; saveFix(); renderFix();
    });
  } else { m.setPosition(pos); }
}
